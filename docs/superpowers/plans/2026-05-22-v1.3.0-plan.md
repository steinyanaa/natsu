```ts
async function searchRssFeed(query: string, feedUrl: string, sourceName: string): Promise<OnlineBookResult[]> {
  const resolvedUrl = feedUrl.replace(/\{q\}/g, encodeURIComponent(query));
  try {
    const response = await withTimeout(net.fetch(resolvedUrl, {
      headers: { "Accept": "application/rss+xml, application/atom+xml, application/xml, text/xml" }
    }), 15000, null);
    if (!response || !response.ok) return [];
    const xml = await response.text();
    const doc = parse(xml);
    const lowerQuery = query.toLowerCase();

    // Support both RSS <item> and Atom <entry>
    const items = doc.querySelectorAll("item, entry");
    const results: OnlineBookResult[] = [];

    for (const item of items) {
      const title = item.querySelector("title")?.text?.trim() ?? "";
      const author = item.querySelector("author name, dc\\:creator, author")?.text?.trim() ?? "";
      const link = item.querySelector("enclosure")?.getAttribute("url")
        ?? item.querySelector("link[rel='enclosure']")?.getAttribute("href")
        ?? item.querySelector("link")?.getAttribute("href")
        ?? item.querySelector("link")?.text?.trim()
        ?? "";

      // filter by query if the URL doesn't contain a query placeholder
      if (!feedUrl.includes("{q}")) {
        const haystack = `${title} ${author}`.toLowerCase();
        if (!haystack.includes(lowerQuery)) continue;
      }

      if (!link) continue;

      const enclosureType = item.querySelector("enclosure")?.getAttribute("type") ?? "";
      const format: BookFormat | undefined =
        link.endsWith(".epub") || enclosureType.includes("epub") ? "epub"
        : link.endsWith(".pdf") || enclosureType.includes("pdf") ? "pdf"
        : link.endsWith(".txt") || enclosureType.includes("text/plain") ? "txt"
        : undefined;

      if (!format) continue;

      const coverUrl = item.querySelector("image url, media\\:thumbnail, itunes\\:image")?.text?.trim()
        ?? item.querySelector("media\\:thumbnail")?.getAttribute("url")
        ?? undefined;

      results.push({
        id: `rss-${Buffer.from(link).toString("base64").slice(0, 16)}`,
        source: sourceName,
        title: title || "Untitled",
        author: author || undefined,
        language: undefined,
        subjects: [],
        coverUrl: coverUrl || undefined,
        downloadUrl: link,
        format,
        sizeLabel: undefined
      });
    }
    return results.slice(0, 40);
  } catch {
    return [];
  }
}
```

```ts
    if (typeof bookId !== "string" || !bookId) return false;
    const books = store.get("books", []);
    const book = books.find((b) => b.id === bookId);
    if (!book) return false;

    // Already has a cover on disk — skip
    try { await fs.access(coverPathFor(bookId)); return true; } catch { /* proceed */ }

    const isbn = (() => {
      // Try to extract from title/author heuristic — real ISBN parsing happens in Task 5
      return undefined as string | undefined;
    })();

    let imageUrl: string | undefined;

    // Strategy 1: OpenLibrary by ISBN
    if (isbn) {
      imageUrl = `https://covers.openlibrary.org/b/isbn/${isbn}-L.jpg`;
    }

    // Strategy 2: Google Books by title+author
    if (!imageUrl) {
      try {
        const q = encodeURIComponent(`intitle:${book.title}${book.author ? `+inauthor:${book.author}` : ""}`);
        const resp = await withTimeout(net.fetch(`https://www.googleapis.com/books/v1/volumes?q=${q}&maxResults=1`), 8000, null);
        if (resp?.ok) {
          const data = await resp.json() as { items?: Array<{ volumeInfo?: { imageLinks?: { thumbnail?: string } } }> };
          const thumb = data?.items?.[0]?.volumeInfo?.imageLinks?.thumbnail;
          if (thumb) {
            imageUrl = thumb.replace("http://", "https://").replace("&zoom=1", "&zoom=3");
          }
        }
      } catch { /* ignore */ }
    }

    if (!imageUrl) return false;

    try {
      const imgResp = await withTimeout(net.fetch(imageUrl), 8000, null);
      if (!imgResp?.ok) return false;
      const buf = Buffer.from(await imgResp.arrayBuffer());
      if (buf.length < 2000) return false; // placeholder / empty image guard
      await ensureCoverDir();
      await fs.writeFile(coverPathFor(bookId), buf);
      return true;
    } catch {
      return false;
    }
  });
```

```ts
  ipcMain.handle("library:getSessionsByDate", () => {
    const books = store.get("books", []);
    const dayMap = new Map<string, number>();
    for (const book of books) {
      for (const session of book.readingSessions ?? []) {
        const day = session.start.slice(0, 10);
        const mins = (new Date(session.end).getTime() - new Date(session.start).getTime()) / 60000;
        dayMap.set(day, (dayMap.get(day) ?? 0) + mins);
      }
    }
    return [...dayMap.entries()]
      .map(([date, minutes]) => ({ date, minutes: Math.round(minutes) }))
      .sort((a, b) => a.date.localeCompare(b.date));
  });
```

```tsx
import type * as React from "react";
import { useState } from "react";
import { Check, Plus } from "lucide-react";
import type { BookSourcePreset, OnlineSource } from "../types";
import { BOOK_SOURCE_PRESETS } from "./presets";

function defaultId(): string {
  return `preset-${Date.now()}`;
}

export function PresetsRow({
  currentSources,
  onAdd
}: {
  currentSources: OnlineSource[];
  onAdd: (source: OnlineSource) => void;
}) {
  const [promptId, setPromptId] = useState<string | null>(null);
  const [urlDraft, setUrlDraft] = useState("");

  const isActive = (preset: BookSourcePreset) =>
    currentSources.some((s) => s.id === preset.id || (s.name === preset.name && s.kind === preset.kind));

  const handleChipClick = (preset: BookSourcePreset) => {
    if (isActive(preset)) return;
    if (preset.requiresUserUrl) {
      setPromptId(preset.id);
      setUrlDraft("");
    } else {
      onAdd({
        id: preset.id,
        name: preset.name,
        enabled: true,
        kind: preset.kind,
        value: preset.value
      });
    }
  };

  const confirmUrl = (preset: BookSourcePreset) => {
    if (!urlDraft.trim()) return;
    onAdd({
      id: defaultId(),
      name: preset.name,
      enabled: true,
      kind: preset.kind,
      value: urlDraft.trim()
    });
    setPromptId(null);
    setUrlDraft("");
  };

  return (
    <div className="presets-row">
      <p className="presets-label">预设书源</p>
      <div className="presets-chips">
        {BOOK_SOURCE_PRESETS.map((preset) => {
          const active = isActive(preset);
          return (
            <button
              key={preset.id}
              type="button"
              title={preset.description}
              className={`preset-chip${active ? " active" : ""}`}
              onClick={() => handleChipClick(preset)}
            >
              {active ? <Check size={13} /> : <Plus size={13} />}
              {preset.name}
            </button>
          );
        })}
      </div>
      {promptId && (() => {
        const preset = BOOK_SOURCE_PRESETS.find((p) => p.id === promptId)!;
        return (
          <div className="preset-url-prompt">
            <p className="preset-url-hint">{preset.description} — 请粘贴镜像地址：</p>
            <div className="preset-url-row">
              <input
                className="source-input"
                placeholder="https://..."
                value={urlDraft}
                onChange={(e) => setUrlDraft(e.target.value)}
                onKeyDown={(e) => {
                  if (e.key === "Enter") confirmUrl(preset);
                  if (e.key === "Escape") setPromptId(null);
                }}
                autoFocus
              />
              <button type="button" className="btn-primary small" onClick={() => confirmUrl(preset)}>
                添加
              </button>
              <button type="button" className="btn-ghost small" onClick={() => setPromptId(null)}>
                取消
              </button>
            </div>
          </div>
        );
      })()}
    </div>
  );
}
```

```ts
import type { BookRecord, Highlight } from "../types";

function chapterLabel(highlight: Highlight, index: number): string {
  return `第 ${index + 1} 章`;
}

function groupByChapter(highlights: Highlight[]): Map<string, Highlight[]> {
  const map = new Map<string, Highlight[]>();
  for (const h of highlights) {
    const key = h.chapterId;
    if (!map.has(key)) map.set(key, []);
    map.get(key)!.push(h);
  }
  return map;
}

export function exportMarkdown(book: BookRecord): string {
  const highlights = book.highlights ?? [];
  if (!highlights.length) return `# ${book.title} — 笔记\n\n（暂无高亮或笔记）\n`;

  const grouped = groupByChapter(highlights);
  const lines: string[] = [`# ${book.title} — 笔记`, "", `作者：${book.author ?? "未知"}`, ""];
  let chapterIndex = 0;

  for (const [, items] of grouped) {
    lines.push(`## ${chapterLabel(items[0], chapterIndex)}`);
    lines.push("");
    for (const h of items) {
      lines.push(`> ${h.selectedText}`);
      lines.push("");
      if (h.note?.trim()) {
        lines.push(`*${h.note.trim()}*`);
        lines.push("");
      }
      lines.push("---");
      lines.push("");
    }
    chapterIndex++;
  }

  return lines.join("\n");
}

export function exportAnkiTsv(book: BookRecord): string {
  const highlights = book.highlights ?? [];
  const rows = highlights.map((h) => {
    const front = h.selectedText.replace(/\t/g, " ").replace(/\n/g, " ");
    const back = [
      h.note?.trim() ?? "",
      `《${book.title}》${book.author ? ` · ${book.author}` : ""}`
    ].filter(Boolean).join("<br>");
    return `${front}\t${back}`;
  });
  return rows.join("\n");
```

```tsx
import type * as React from "react";
```

```tsx
const CELL = 11;   // px per cell
const GAP  = 2;    // px gap between cells
const STEP = CELL + GAP;
const WEEKS = 52;

function bucketColor(minutes: number): string {
  if (minutes <= 0) return "var(--md-sys-color-surface-container)";
  if (minutes < 16)  return "color-mix(in srgb, var(--md-sys-color-primary) 25%, transparent)";
  if (minutes < 46)  return "color-mix(in srgb, var(--md-sys-color-primary) 55%, transparent)";
  return "color-mix(in srgb, var(--md-sys-color-primary) 90%, transparent)";
}

function isoDate(date: Date): string {
  return date.toISOString().slice(0, 10);
}

export function HeatmapCalendar({ stats }: { stats: DailyReadingStat[] }) {
  const statMap = useMemo(() => {
    const m = new Map<string, number>();
    for (const s of stats) m.set(s.date, s.minutes);
    return m;
  }, [stats]);

  // Build 52 columns, each column = one week (Sun→Sat)
  const today = new Date();
  const todayDay = today.getDay(); // 0=Sun

  // Start of the grid: go back 52 weeks from the most recent Sunday
  const startDate = new Date(today);
  startDate.setDate(today.getDate() - todayDay - (WEEKS - 1) * 7);

  const columns: Array<Array<{ date: string; minutes: number }>> = [];
  for (let w = 0; w < WEEKS; w++) {
    const col: Array<{ date: string; minutes: number }> = [];
    for (let d = 0; d < 7; d++) {
      const cell = new Date(startDate);
      cell.setDate(startDate.getDate() + w * 7 + d);
      const dateStr = isoDate(cell);
      col.push({ date: dateStr, minutes: statMap.get(dateStr) ?? 0 });
    }
    columns.push(col);
  }

  const svgW = WEEKS * STEP;
  const svgH = 7 * STEP;

  return (
```

```tsx
import type * as React from "react";
import { useMemo } from "react";
import type { DailyReadingStat } from "../types";

const W = 440;
const H = 80;
const PAD_L = 8;
const PAD_R = 8;
const PAD_T = 8;
const PAD_B = 20;

function cubicBezier(points: Array<[number, number]>): string {
  if (points.length < 2) return "";
  let d = `M${points[0][0]},${points[0][1]}`;
  for (let i = 1; i < points.length; i++) {
    const [x0, y0] = points[i - 1];
    const [x1, y1] = points[i];
    const cpX = (x0 + x1) / 2;
    d += ` C${cpX},${y0} ${cpX},${y1} ${x1},${y1}`;
  }
  return d;
}

function lastNDays(stats: DailyReadingStat[], n: number): DailyReadingStat[] {
  const result: DailyReadingStat[] = [];
  const today = new Date();
  for (let i = n - 1; i >= 0; i--) {
    const d = new Date(today);
    d.setDate(today.getDate() - i);
    const date = d.toISOString().slice(0, 10);
    result.push({ date, minutes: stats.find((s) => s.date === date)?.minutes ?? 0 });
  }
  return result;
}

export function ReadingCurve({ stats }: { stats: DailyReadingStat[] }) {
  const days = useMemo(() => lastNDays(stats, 30), [stats]);
  const maxMin = useMemo(() => Math.max(1, ...days.map((d) => d.minutes)), [days]);

  const chartW = W - PAD_L - PAD_R;
  const chartH = H - PAD_T - PAD_B;

  const points: Array<[number, number]> = days.map((d, i) => [
    PAD_L + (i / (days.length - 1)) * chartW,
    PAD_T + chartH - (d.minutes / maxMin) * chartH
  ]);

  const path = cubicBezier(points);
  const fillPath = path + ` L${points[points.length - 1][0]},${PAD_T + chartH} L${points[0][0]},${PAD_T + chartH} Z`;

  // Label every 5th day
  const labels = days.filter((_, i) => i % 5 === 0 || i === days.length - 1);

  return (
    <div className="reading-curve-wrap">
      <svg
        viewBox={`0 0 ${W} ${H}`}
        className="reading-curve-svg"
        role="img"
        aria-label="近 30 天阅读趋势"
        preserveAspectRatio="none"
      >
        <defs>
          <linearGradient id="curveGrad" x1="0" y1="0" x2="0" y2="1">
            <stop offset="0%" stopColor="var(--md-sys-color-primary)" stopOpacity="0.22" />
            <stop offset="100%" stopColor="var(--md-sys-color-primary)" stopOpacity="0.02" />
          </linearGradient>
        </defs>
        <path d={fillPath} fill="url(#curveGrad)" />
        <path d={path} fill="none" stroke="var(--md-sys-color-primary)" strokeWidth="2" strokeLinecap="round" />
        {labels.map((d) => {
          const i = days.findIndex((s) => s.date === d.date);
          const x = PAD_L + (i / (days.length - 1)) * chartW;
          return (
            <text
              key={d.date}
              x={x}
              y={H - 4}
              fontSize="9"
              textAnchor="middle"
              fill="var(--reader-muted)"
            >
```

```ts
/**
 * Rolling reading-speed calculator.
 * Maintained as a module-level singleton so TextPane can call it on each page turn.
 * The weekly average is fetched separately via the reading sessions data.
 */

const SAMPLE_CAP = 10;

interface SpeedSample {
  chars: number;
  ms: number;
}

const samples: SpeedSample[] = [];

/**
 * Record a page turn event.
 * @param chars  Number of characters visible on the page that just completed
 * @param ms     Time elapsed since the page was displayed (milliseconds)
 */
export function recordPageTurn(chars: number, ms: number): void {
  if (chars <= 0 || ms <= 0) return;
  samples.push({ chars, ms });
  if (samples.length > SAMPLE_CAP) samples.shift();
}

/**
 * Current session WPM (characters per minute).
 * Returns 0 if no samples yet.
 */
export function sessionCpm(): number {
  if (!samples.length) return 0;
  const totalChars = samples.reduce((s, x) => s + x.chars, 0);
  const totalMs = samples.reduce((s, x) => s + x.ms, 0);
  return totalMs > 0 ? Math.round((totalChars / totalMs) * 60_000) : 0;
}

export function clearSamples(): void {
  samples.length = 0;
}
```

```ts
import { useMemo, useState } from "react";
import type { BookFormat, BookRecord, Collection } from "../types";

type ShelfFilter = "all" | "novels" | "comics" | "pdf";
type ShelfView = "grid" | "list";
type ShelfSort = "recent" | "title" | "author" | "progress" | "size";

const novelFormats: BookFormat[] = ["epub", "txt", "mobi", "azw3"];
const comicFormats: BookFormat[] = ["cbz", "zip", "cbr", "rar"];

function bookActivityTime(book: BookRecord): number {
  return new Date(book.lastOpenedAt ?? book.progress?.updatedAt ?? book.importedAt).getTime();
}

export interface UseBookShelfReturn {
  query: string;
  filter: ShelfFilter;
  view: ShelfView;
  sort: ShelfSort;
  filteredBooks: BookRecord[];
  setQuery: React.Dispatch<React.SetStateAction<string>>;
  setFilter: React.Dispatch<React.SetStateAction<ShelfFilter>>;
  setView: React.Dispatch<React.SetStateAction<ShelfView>>;
  setSort: React.Dispatch<React.SetStateAction<ShelfSort>>;
}

export function useBookShelf(
  books: BookRecord[],
  collections: Collection[],
  activeCollectionId: string | null
): UseBookShelfReturn {
  const [query, setQuery] = useState("");
  const [filter, setFilter] = useState<ShelfFilter>("all");
  const [view, setView] = useState<ShelfView>("grid");
  const [sort, setSort] = useState<ShelfSort>("recent");

  const filteredBooks = useMemo(() => {
    let result = books;

    if (activeCollectionId) {
      const col = collections.find((c) => c.id === activeCollectionId);
      const ids = new Set(col?.bookIds ?? []);
      result = result.filter((b) => ids.has(b.id));
    }

    if (filter === "novels") result = result.filter((b) => novelFormats.includes(b.format));
    else if (filter === "comics") result = result.filter((b) => comicFormats.includes(b.format));
    else if (filter === "pdf") result = result.filter((b) => b.format === "pdf");

    if (query.trim()) {
      const q = query.toLowerCase();
      result = result.filter(
        (b) => b.title.toLowerCase().includes(q) || b.author?.toLowerCase().includes(q)
      );
    }

    return [...result].sort((a, b) => {
      switch (sort) {
        case "title": return a.title.localeCompare(b.title);
        case "author": return (a.author ?? "").localeCompare(b.author ?? "");
        case "progress": return (b.progress?.percent ?? 0) - (a.progress?.percent ?? 0);
        case "size": return b.size - a.size;
        default: return bookActivityTime(b) - bookActivityTime(a);
      }
    });
  }, [books, collections, activeCollectionId, filter, query, sort]);

  return { query, filter, view, sort, filteredBooks, setQuery, setFilter, setView, setSort };
}
```

Extract the chapter-loading logic. This hook encapsulates `document`, `error`, `load()`, and related state:

```ts
import { useEffect, useRef, useState } from "react";
import { parseEpubDocument } from "../readers/epub";
import { parseMobiDocument } from "../readers/mobi";
import { parseTxtDocument } from "../readers/text";
import type { BookRecord, ParsedTextDocument, TocItem } from "../types";

export interface UseEpubChapterReturn {
  document: ParsedTextDocument | undefined;
  error: string;
}

export function useEpubChapter(
  book: BookRecord,
  parser: "txt" | "mobi" | "epub",
  onToc: (toc: TocItem[]) => void,
  onChapters?: (chapters: ParsedTextDocument["chapters"]) => void,
  errorLabel?: string
): UseEpubChapterReturn {
  const [document, setDocument] = useState<ParsedTextDocument | undefined>();
  const [error, setError] = useState("");

  useEffect(() => {
    let cancelled = false;
    let ownedUrls: string[] = [];

    async function load() {
      setDocument(undefined);
      setError("");
      try {
        const buffer = await fetch(book.fileUrl).then((r) => r.arrayBuffer());
        const parsed =
          parser === "txt"
            ? parseTxtDocument(buffer, book.title)
            : parser === "mobi"
              ? await parseMobiDocument(buffer, book.title)
              : await parseEpubDocument(new Blob([buffer]), book.title);
        ownedUrls = parsed.objectUrls ?? [];
        if (cancelled) {
          ownedUrls.forEach((url) => URL.revokeObjectURL(url));
          return;
        }
        setDocument(parsed);
        onToc(parsed.toc);
        onChapters?.(parsed.chapters);
      } catch {
        if (!cancelled) setError(errorLabel ?? "unsupported");
      }
    }

    void load();
    return () => {
      cancelled = true;
      ownedUrls.forEach((url) => URL.revokeObjectURL(url));
    };
  }, [book.fileUrl, book.title, onToc, onChapters, parser, errorLabel]);

  return { document, error };
}
```
